Class Transaction
    attribute id
    attribute merchant
    attribute date  
    attribute amount
    constructor(id, merchant, date, amount)

Main
    transactions = new List<Transaction>()
    transactions.add(new Transaction(1, "Amazon", "2025-01-12", 1200))
    transactions.add(new Transaction(2, "Flipkart", "2025-01-15", 800))
    transactions.add(new Transaction(3, "Amazon", "2025-02-03", 2000))
    transactions.add(new Transaction(4, "Myntra", "2025-02-10", 1500))
    transactions.add(new Transaction(5, "Amazon", "2025-01-20", 600))

    
    report = transactions.stream()
        .collect(
            groupingBy(
                txn -> (txn.merchant + "-" + extractMonth(txn.date)),
                summingDouble(txn -> txn.amount)
            )
        )

        for each entry in report
        print "Merchant & Month: " + entry.key
        print "Total Amount: " + entry.value

Function extractMonth(date)
    return substring(date, 0, 7)  // "YYYY-MM"
